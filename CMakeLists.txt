cmake_minimum_required(VERSION 3.16)
project(nicolai_native_port LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(nicolai_port STATIC
    src/voice_db.cpp src/edat.cpp src/address_space.cpp src/assets.cpp src/stateful_tds.cpp
    src/huffman.cpp src/ana_index.cpp src/cmp16.cpp src/cmp16_decode.cpp src/cmp16_codebook.cpp src/static_files.cpp src/legacy_tds.cpp src/legacy_phone_features.cpp src/legacy_runtime_state.cpp src/legacy_runtime_grain_m36.cpp src/legacy_runtime_cross_m36.cpp src/legacy_runtime_executor_m36.cpp src/legacy_window_m36.cpp
    src/resource_registry.cpp src/voice_catalog.cpp src/diphone_catalog.cpp src/g711.cpp src/psola_join.cpp src/td_psola.cpp src/seg_schedule.cpp src/hybrid_psola.cpp src/russian_stress.cpp src/russian_legacy.cpp src/russian_frontend.cpp src/legacy_prosody.cpp src/pc_parity.cpp src/engine.cpp src/wav_writer.cpp)
target_include_directories(nicolai_port PUBLIC include)
if(NOT ANDROID)
  add_executable(nicolai_m33_probe tools/nicolai_m33_probe.cpp)
  target_link_libraries(nicolai_m33_probe PRIVATE nicolai_port)
  add_executable(nicolai_m34_probe tools/nicolai_m34_probe.cpp)
  target_link_libraries(nicolai_m34_probe PRIVATE nicolai_port)
  add_executable(nicolai_m36_phone_probe tools/nicolai_m36_phone_probe.cpp)
  target_link_libraries(nicolai_m36_phone_probe PRIVATE nicolai_port)
  add_executable(nicolai_m36_route_probe tools/nicolai_m36_route_probe.cpp)
  target_link_libraries(nicolai_m36_route_probe PRIVATE nicolai_port)
  if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_executable(nicolai_m36_runtime_capture tools/nicolai_m36_runtime_capture.cpp)
    add_executable(nicolai_legacy_startup_trace tools/nicolai_legacy_startup_trace.cpp)
  endif()
  foreach(tool nicolai_probe nicolai_m2_probe nicolai_m3_probe nicolai_m4_probe nicolai_m5_probe nicolai_m8_probe nicolai_m10_probe nicolai_m11_probe nicolai_m12_probe nicolai_m13_probe nicolai_m14_probe nicolai_m15_probe nicolai_m16_probe nicolai_m17_probe nicolai_m18_probe nicolai_m26_probe nicolai_m27_probe nicolai_render nicolai_batch_render)
    add_executable(${tool} tools/${tool}.cpp)
    target_link_libraries(${tool} PRIVATE nicolai_port)
  endforeach()
endif()
if(ANDROID)
  add_library(nicolai_native SHARED android/jni/nicolai_jni.cpp)
  target_link_libraries(nicolai_native PRIVATE nicolai_port log)
  target_include_directories(nicolai_native PRIVATE include)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_executable(m36_route_capture_test tests/m36_route_capture_test.cpp)
    add_test(NAME m36_route_capture_test COMMAND m36_route_capture_test)
    add_executable(legacy_startup_trace_fixture tests/legacy_startup_trace_fixture.cpp)
  endif()
  foreach(test huffman_test ana_index_test cmp16_test static_files_test address_space_test resource_registry_test voice_catalog_test diphone_catalog_test cmp16_decode_test cmp16_codebook_test g711_test psola_join_test td_psola_test seg_schedule_test hybrid_psola_test russian_stress_test russian_legacy_test russian_frontend_test legacy_prosody_test pc_parity_test legacy_phone_features_test legacy_runtime_state_test legacy_runtime_grain_m36_test legacy_runtime_cross_m36_test legacy_runtime_executor_m36_test legacy_window_m36_test stateful_tds_m36_test)
    add_executable(${test} tests/${test}.cpp)
    target_link_libraries(${test} PRIVATE nicolai_port)
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  add_test(NAME nicolai_m36_phone_probe_contract COMMAND nicolai_m36_phone_probe)
endif()
